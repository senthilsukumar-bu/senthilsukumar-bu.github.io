spark = SparkSession.builder.appName("CareerCompassData").getOrCreate()

from pathlib import Path

data_dir = Path("./Jobs_2026")  
parquet_files = sorted(data_dir.glob("*.parquet"))

df = spark.read.parquet(*[str(path) for path in parquet_files])

model_source_cols = [
    "SALARY_TO", "SALARY_FROM", "SALARY",
    "MIN_YEARS_EXPERIENCE", "MAX_YEARS_EXPERIENCE", "DURATION",
    "EDUCATION_LEVELS_NAME", "EMPLOYMENT_TYPE_NAME", "REMOTE_TYPE_NAME",
    "IS_INTERNSHIP", "COMPANY_IS_STAFFING"
]
df = df.select(*model_source_cols)

# Cast numeric columns to double and replace NaN with null
numeric_cols = ["SALARY_TO", "SALARY_FROM", "SALARY",
                "MIN_YEARS_EXPERIENCE", "MAX_YEARS_EXPERIENCE", "DURATION"]

for c in numeric_cols:
    df = df.withColumn(c, F.expr("try_cast(`" + c + "` as double)"))
    df = df.withColumn(
        c,
        F.when(F.col(c).isNull() | F.isnan(F.col(c)), None).otherwise(F.col(c))
    )

# REMOTE_TYPE_NAME and EDUCATION_LEVELS_NAME — ArrayType: explode directly
for col_name in ["REMOTE_TYPE_NAME", "EDUCATION_LEVELS_NAME"]:
    if isinstance(df.schema[col_name].dataType, ArrayType):
        df = df.withColumn(col_name, F.explode_outer(F.col(col_name)))

# Normalize REMOTE_TYPE_NAME
df = df.withColumn(
    "REMOTE_TYPE_NAME",
    F.when(F.col("REMOTE_TYPE_NAME") == "On-site", "Onsite")
     .when(F.col("REMOTE_TYPE_NAME").isNull() | (F.col("REMOTE_TYPE_NAME") == "Unknown"), "Unknown")
     .otherwise(F.col("REMOTE_TYPE_NAME"))
)

# EMPLOYMENT_TYPE_NAME — comma-separated string: split then explode
df = df.withColumn(
    "EMPLOYMENT_TYPE_NAME",
    F.explode_outer(F.split(F.col("EMPLOYMENT_TYPE_NAME"), r",\s*"))
)
df = df.withColumn("EMPLOYMENT_TYPE_NAME", F.trim(F.col("EMPLOYMENT_TYPE_NAME")))

# Normalize EMPLOYMENT_TYPE_NAME — collapse all variations to canonical labels
e = F.lower(F.col("EMPLOYMENT_TYPE_NAME"))
df = df.withColumn(
    "EMPLOYMENT_TYPE_NAME",
    F.when(e.contains("intern"),                           "Internship")
     .when(e.contains("per diem") | e.contains("perdiem"), "Per Diem")
     .when(e.contains("temp"),                             "Temporary")
     .when(e.contains("contract"),                         "Contract")
     .when(e.contains("part"),                             "Part Time")
     .when(e.contains("permanent"),                        "Permanent")
     .when(e.contains("remote"),                           "Remote")
     .when(e.contains("full"),                             "Full Time")
     .otherwise(None)
)

# Replace empty strings with null across all string columns
for col_name in ["REMOTE_TYPE_NAME", "EDUCATION_LEVELS_NAME", "EMPLOYMENT_TYPE_NAME"]:
    df = df.withColumn(
        col_name,
        F.when(F.trim(F.col(col_name)) == "", None).otherwise(F.col(col_name))
    )

from pyspark.ml import Pipeline
from pyspark.ml.feature import StringIndexer, OneHotEncoder, VectorAssembler
from pyspark.sql import SparkSession
from pyspark.sql import functions as F

# ── Feature selection ─────────────────────────────────────────────────────────
continuous_cols = [
    "SALARY_FROM",
    "MAX_YEARS_EXPERIENCE",
    "DURATION",
    "MIN_YEARS_EXPERIENCE",
]
categorical_cols = [
    "REMOTE_TYPE_NAME",
    "EMPLOYMENT_TYPE_NAME",
    "EDUCATION_LEVELS_NAME",
]
target_col = "SALARY_FROM"   # ← use SALARY_FROM (cleaner numeric) as target

# ── 1. Cast all continuous cols to double (tolerate bad strings) ──────────────
df_clean = df
for c in continuous_cols:
    df_clean = df_clean.withColumn(c, F.expr(f"try_cast(`{c}` as double)"))

# ── 2. Replace empty strings in categoricals ──────────────────────────────────
for c in categorical_cols:
    df_clean = df_clean.withColumn(
        c, F.when(F.trim(F.col(c)) == "", None).otherwise(F.col(c))
    )

# ── 3. Only require target + MIN_YEARS_EXPERIENCE to be non-null ──────────────
df_clean = df_clean.dropna(subset=[target_col, "MIN_YEARS_EXPERIENCE"])

print(f"Rows in raw data     : {df.count():,}")
print(f"Rows after cleaning  : {df_clean.count():,}")

# ── StringIndexer + OneHotEncoder ─────────────────────────────────────────────
indexers = [
    StringIndexer(inputCol=c, outputCol=f"{c}_idx", handleInvalid="keep")
    for c in categorical_cols
]
encoders = [
    OneHotEncoder(inputCol=f"{c}_idx", outputCol=f"{c}_ohe")
    for c in categorical_cols
]
ohe_cols = [f"{c}_ohe" for c in categorical_cols]

# ── VectorAssembler (handleInvalid="keep" fills missing features with 0) ──────
assembler = VectorAssembler(
    inputCols=continuous_cols + ohe_cols,
    outputCol="features",
    handleInvalid="keep"
)

# ── Pipeline ──────────────────────────────────────────────────────────────────
pipeline_model = Pipeline(stages=indexers + encoders + [assembler]).fit(df_clean)
df_featured = pipeline_model.transform(df_clean)

# ── Polynomial feature: MIN_YEARS_EXPERIENCE² ─────────────────────────────────
df_featured = df_featured.withColumn(
    "MIN_YEARS_EXPERIENCE_SQ",
    F.pow(F.col("MIN_YEARS_EXPERIENCE"), 2)
)

# ── Assemble polynomial features ──────────────────────────────────────────────
poly_assembler = VectorAssembler(
    inputCols=continuous_cols + ["MIN_YEARS_EXPERIENCE_SQ"] + ohe_cols,
    outputCol="features_poly",
    handleInvalid="keep"
)
df_poly = poly_assembler.transform(df_featured)

# ── Train/test split ──────────────────────────────────────────────────────────
train_df, test_df = df_poly.randomSplit([0.8, 0.2], seed=42)
print(f"Training rows : {train_df.count():,}")
print(f"Testing  rows : {test_df.count():,}")

# ── Show final structure ──────────────────────────────────────────────────────
show_cols = [target_col] + continuous_cols + categorical_cols + \
            ["MIN_YEARS_EXPERIENCE_SQ", "features", "features_poly"]

df_poly.select(*show_cols).printSchema()

from pyspark.ml.regression import LinearRegression
from pyspark.ml.evaluation import RegressionEvaluator
from pyspark.sql.functions import udf, col, isnan
from pyspark.sql.types import BooleanType
from scipy import stats
import numpy as np
import pandas as pd
import math

boolean_cols = ["IS_INTERNSHIP", "COMPANY_IS_STAFFING"]

# ── NaN guard on feature vector ───────────────────────────────────────────────
@udf(BooleanType())
def has_nan_or_inf(v):
    if v is None:
        return True
    return any(math.isnan(x) or math.isinf(x) for x in v.toArray())

train_clean = train_df.filter(~has_nan_or_inf(col("features")))
test_clean  = test_df.filter(~has_nan_or_inf(col("features")))

print("Train rows :", train_clean.count())
print("Test  rows :", test_clean.count())



# ── Train ─────────────────────────────────────────────────────────────────────
lr = LinearRegression(
    featuresCol="features",
    labelCol=target_col,
    maxIter=100,
    solver="normal",
    regParam=0.0,
    elasticNetParam=0.0,
    standardization=False
)
lr_model = lr.fit(train_clean)

# ── Evaluate ──────────────────────────────────────────────────────────────────
predictions = lr_model.transform(test_clean)

evaluator = RegressionEvaluator(labelCol=target_col, predictionCol="prediction")
r2   = evaluator.setMetricName("r2").evaluate(predictions)
rmse = evaluator.setMetricName("rmse").evaluate(predictions)
mae  = evaluator.setMetricName("mae").evaluate(predictions)

print("Intercept :", round(lr_model.intercept, 4))
print("R2        :", round(r2, 4))
print("RMSE      :", round(rmse, 2))
print("MAE       :", round(mae, 2))

# ── SE / t-values / p-values ──────────────────────────────────────────────────
summary = lr_model.summary
coefs   = np.array(lr_model.coefficients.toArray())
n_feats = len(coefs)

try:
    se    = np.array(summary.coefficientStandardErrors[:-1])
    tvals = np.array(summary.tValues[:-1])
    pvals = np.array(summary.pValues[:-1])
    se_method = "exact"
except Exception:
    train_pred = lr_model.transform(train_clean)
    n   = train_pred.count()
    rss_row = train_pred.selectExpr(
        "sum(pow(" + target_col + " - prediction, 2)) as rss"
    ).collect()[0]
    rss = rss_row["rss"]
    mse = rss / (n - n_feats - 1)
    se    = np.full(n_feats, np.sqrt(mse / n))
    tvals = coefs / se
    pvals = 2 * (1 - stats.t.cdf(np.abs(tvals), df=n - n_feats - 1))
    se_method = "approximated"

# ── Feature names from VectorAssembler metadata (ground truth) ────────────────
meta  = train_df.schema["features"].metadata
attrs = meta.get("ml_attr", {}).get("attrs", {})

name_map = {}
for attr_type, attr_list in attrs.items():
    for attr in attr_list:
        name_map[attr["idx"]] = attr["name"]

feature_names = [name_map.get(i, "feature_" + str(i)) for i in range(n_feats)]

n_names = len(feature_names)
if n_names != n_feats:
    raise ValueError("Mismatch: " + str(n_names) + " names vs " + str(n_feats) + " features")

print("Feature names:", feature_names)
# ── Coefficient DataFrame ─────────────────────────────────────────────────────
ci_lower = coefs - 1.96 * se
ci_upper = coefs + 1.96 * se

coef_df = pd.DataFrame({
    "Feature"     : feature_names,
    "Coefficient" : coefs.round(4),
    "Std Error"   : se.round(4),
    "t-value"     : tvals.round(4),
    "p-value"     : pvals.round(4),
    "CI Lower"    : ci_lower.round(4),
    "CI Upper"    : ci_upper.round(4),
    "Significant" : ["Yes" if p < 0.05 else "No" for p in pvals],
})

top15 = coef_df.reindex(
    coef_df["Coefficient"].abs().sort_values(ascending=False).index
).head(15)

print("  MAE  =", round(mae, 2), "-> median absolute prediction error")

print("SE method:", se_method)

# AIC  = n·ln(RSS/n) + 2·(k+1)       where k = n_coefficients, +1 for intercept
# BIC  = n·ln(RSS/n) + (k+1)·ln(n)

def compute_aic_bic(model, train_data, label_col):
    n   = train_data.count()
    k   = len(model.coefficients) + 1
    rss = (
        model.transform(train_data)
             .selectExpr(
                 "sum(pow(`" + label_col + "` - prediction, 2)) as rss"
             )
             .collect()[0]["rss"]
    )
    aic = n * math.log(rss / n) + 2 * k
    bic = n * math.log(rss / n) + k * math.log(n)
    return round(aic, 2), round(bic, 2)
