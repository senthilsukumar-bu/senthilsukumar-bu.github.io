Feature Engineering and Train/Test split

from pyspark.ml import Pipeline
from pyspark.ml.feature import StringIndexer, OneHotEncoder, VectorAssembler
from pyspark.sql import SparkSession
from pyspark.sql import functions as F

# ── Feature selection ─────────────────────────────────────────────────────────
continuous_cols = [
    "SALARY_FROM",
    "MAX_YEARS_EXPERIENCE",
    "DURATION",
    "MIN_YEARS_EXPERIENCE",
]
categorical_cols = [
    "REMOTE_TYPE_NAME",
    "EMPLOYMENT_TYPE_NAME",
]
target_col = "SALARY_FROM"   # ← use SALARY_FROM (cleaner numeric) as target

# ── 1. Cast all continuous cols to double (tolerate bad strings) ──────────────
df_clean = df
for c in continuous_cols:
    df_clean = df_clean.withColumn(c, F.expr(f"try_cast(`{c}` as double)"))

# ── 2. Replace empty strings in categoricals ──────────────────────────────────
for c in categorical_cols:
    df_clean = df_clean.withColumn(
        c, F.when(F.trim(F.col(c)) == "", None).otherwise(F.col(c))
    )

# ── 3. Only require target + MIN_YEARS_EXPERIENCE to be non-null ──────────────
df_clean = df_clean.dropna(subset=[target_col, "MIN_YEARS_EXPERIENCE"])

print(f"Rows in raw data     : {df.count():,}")
print(f"Rows after cleaning  : {df_clean.count():,}")

# ── StringIndexer + OneHotEncoder ─────────────────────────────────────────────
indexers = [
    StringIndexer(inputCol=c, outputCol=f"{c}_idx", handleInvalid="keep")
    for c in categorical_cols
]
encoders = [
    OneHotEncoder(inputCol=f"{c}_idx", outputCol=f"{c}_ohe")
    for c in categorical_cols
]
ohe_cols = [f"{c}_ohe" for c in categorical_cols]

# ── VectorAssembler (handleInvalid="keep" fills missing features with 0) ──────
assembler = VectorAssembler(
    inputCols=continuous_cols + ohe_cols,
    outputCol="features",
    handleInvalid="keep"
)

# ── Pipeline ──────────────────────────────────────────────────────────────────
pipeline_model = Pipeline(stages=indexers + encoders + [assembler]).fit(df_clean)
df_featured = pipeline_model.transform(df_clean)

# ── Polynomial feature: MIN_YEARS_EXPERIENCE² ─────────────────────────────────
df_featured = df_featured.withColumn(
    "MIN_YEARS_EXPERIENCE_SQ",
    F.pow(F.col("MIN_YEARS_EXPERIENCE"), 2)
)

# ── Assemble polynomial features ──────────────────────────────────────────────
poly_assembler = VectorAssembler(
    inputCols=continuous_cols + ["MIN_YEARS_EXPERIENCE_SQ"] + ohe_cols,
    outputCol="features_poly",
    handleInvalid="keep"
)
df_poly = poly_assembler.transform(df_featured)

# ── Train/test split ──────────────────────────────────────────────────────────
train_df, test_df = df_poly.randomSplit([0.8, 0.2], seed=42)
print(f"Training rows : {train_df.count():,}")
print(f"Testing  rows : {test_df.count():,}")

# ── Show final structure ──────────────────────────────────────────────────────
show_cols = [target_col] + continuous_cols + categorical_cols + \
            ["MIN_YEARS_EXPERIENCE_SQ", "features", "features_poly"]

df_poly.select(*show_cols).printSchema()
df_poly.select(*show_cols).show(5, truncate=True)

from pyspark.ml.regression import LinearRegression
from pyspark.ml.evaluation import RegressionEvaluator
from pyspark.sql.functions import udf, col
from pyspark.sql.types import BooleanType
from scipy import stats
import numpy as np
import pandas as pd
import math

@udf(BooleanType())
def has_nan_or_inf(v):
    if v is None:
        return True
    return any(math.isnan(x) or math.isinf(x) for x in v.toArray())

train_df = train_df.filter(~has_nan_or_inf(col("features")))
test_df  = test_df.filter(~has_nan_or_inf(col("features")))

print(f"Train rows after NaN filter : {train_df.count():,}")
print(f"Test  rows after NaN filter : {test_df.count():,}")

# ── Train ─────────────────────────────────────────────────────────────────────
lr = LinearRegression(
    featuresCol="features",
    labelCol=target_col,
    maxIter=100,
    solver="normal",
    regParam=0.0,
    elasticNetParam=0.0,
    standardization=False
)
lr_model = lr.fit(train_df)

# ── Evaluate on test set ──────────────────────────────────────────────────────
predictions = lr_model.transform(test_df)

r2   = RegressionEvaluator(labelCol=target_col, predictionCol="prediction", metricName="r2").evaluate(predictions)
rmse = RegressionEvaluator(labelCol=target_col, predictionCol="prediction", metricName="rmse").evaluate(predictions)
mae  = RegressionEvaluator(labelCol=target_col, predictionCol="prediction", metricName="mae").evaluate(predictions)

print(f"Intercept : {lr_model.intercept:,.4f}")
print(f"R²        : {r2:.4f}")
print(f"RMSE      : {rmse:,.2f}")
print(f"MAE       : {mae:,.2f}")

# ── Coefficients, SE, t-values, p-values ─────────────────────────────────────
summary = lr_model.summary
coefs   = np.array(lr_model.coefficients.toArray())
n_feats = len(coefs)

try:
    # Exact SE — only works when X'X is invertible
    se    = np.array(summary.coefficientStandardErrors[:-1])
    tvals = np.array(summary.tValues[:-1])
    pvals = np.array(summary.pValues[:-1])
    se_method = "exact"
except Exception:
    # Fallback: approximate SE from residuals (near-multicollinearity in OHE features)
    train_pred = lr_model.transform(train_df)
    n   = train_pred.count()
    rss = train_pred.selectExpr(f"sum(pow({target_col} - prediction, 2)) as rss").collect()[0]["rss"]
    mse = rss / (n - n_feats - 1)
    se    = np.full(n_feats, np.sqrt(mse / n))
    tvals = coefs / se
    pvals = 2 * (1 - stats.t.cdf(np.abs(tvals), df=n - n_feats - 1))
    se_method = "approximated (X'X near-singular due to OHE multicollinearity)"

# ── Diagnostic check ──────────────────────────────────────────────────────────
print("\n---This is Diagnostic check, No need to print it in the final doc---")
print("---The numbers below should all be same---")
print(f"SE method:          {se_method}")
print(f"Length of features: {n_feats}")
print(f"Length of coefs:    {len(coefs)}")
print(f"Length of se:       {len(se)}")
print(f"Length of tvals:    {len(tvals)}")
print(f"Length of pvals:    {len(pvals)}")

# ── Build feature names ───────────────────────────────────────────────────────
n_ohe = n_feats - len(continuous_cols)
feature_names = list(continuous_cols) + [f"ohe_feature_{i}" for i in range(n_ohe)]

# ── Coefficient summary DataFrame ─────────────────────────────────────────────
ci_lower = coefs - 1.96 * se
ci_upper = coefs + 1.96 * se

coef_df = pd.DataFrame({
    "Feature"     : feature_names,
    "Coefficient" : coefs.round(4),
    "Std Error"   : se.round(4),
    "t-value"     : tvals.round(4),
    "p-value"     : pvals.round(4),
    "CI Lower"    : ci_lower.round(4),
    "CI Upper"    : ci_upper.round(4),
    "Significant" : ["Yes" if p < 0.05 else "No" for p in pvals],
})

print("\nTop 15 features by absolute coefficient:")
top15 = coef_df.reindex(coef_df["Coefficient"].abs().sort_values(ascending=False).index).head(15)
print(top15.to_string(index=False))

# ── Interpretation ────────────────────────────────────────────────────────────
sig_count = (coef_df["Significant"] == "Yes").sum()
print(f"\nModel Performance:")
print(f"  R² = {r2:.4f}  → model explains {r2*100:.1f}% of variance in salary")
print(f"  RMSE = {rmse:,.2f}  → average prediction error in salary units")
print(f"  MAE  = {mae:,.2f}  → median absolute prediction error")
print(f"\n{sig_count} of {n_feats} features are statistically significant (p < 0.05)")
